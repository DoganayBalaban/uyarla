```ts
/** Ücretsiz katman: takvim ayı başına (DOG-158, K-107). */
export const FREE_QUOTA = { analyses: 5, adaptations: 1 } as const

export type Feature = "analysis" | "adaptation" | "interview_prep" | "practice" | "form_answer" | "linkedin_rewrite"
export type DenyCode = "analysis_quota" | "adaptation_quota" | "subscription_required" | "daily_cap"

/** Polar aboneliğinin bizde tutulan hâli; Prisma satırı buna uyuyor. */
export interface SubscriptionLike {
  status: string
  currentPeriodEnd: Date | null
  cancelAtPeriodEnd: boolean
  trialEnd: Date | null
  endedAt: Date | null
  createdAt: Date
}

export interface Usage {
  /** Bu takvim ayında açılan ve başarısız olmayan analizler. */
  analysesThisMonth: number
  /** Bu takvim ayında açılan ve başarısız olmayan uyarlamalar. */
  adaptationsThisMonth: number
  adaptationsLast24h: number
}

export type Decision = { allowed: true; via: "free" | "subscription" | "admin" } | { allowed: false; code: DenyCode }

/**
 * Tek satırın erişim verip vermediği. Polar'da iptal edilen abonelik dönem
 * sonuna kadar `active` + `cancelAtPeriodEnd`; iade ya da iptal edilip biten
 * abonelik `canceled` ve `endedAt` dolu. Bilinmeyen durum erişim vermiyor.
 */
export function grantsAccess(sub: SubscriptionLike, now: Date): boolean {
  if (sub.endedAt && sub.endedAt <= now) return false
  const inPeriod = !!sub.currentPeriodEnd && now < sub.currentPeriodEnd
  switch (sub.status) {
    case "trialing":
    case "active":
      return true
    case "past_due":
      return inPeriod
    case "canceled":
      return inPeriod && sub.cancelAtPeriodEnd
    default:
      return false
  }
}

export function hasAccess(subs: SubscriptionLike[], now: Date): boolean {
  return subs.some((s) => grantsAccess(s, now))
}

/** Ekranda gösterilen abonelik: erişim veren, yoksa en yenisi. */
export function currentSubscription<T extends SubscriptionLike>(subs: T[], now: Date): T | null {
  const newest = [...subs].sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime())
  return newest.find((s) => grantsAccess(s, now)) ?? newest[0] ?? null
}

/** Deneme hesap başına bir kez: hiç aboneliği olmamış kullanıcıya. */
export function trialEligible(subs: SubscriptionLike[]): boolean {
  return subs.length === 0
}

/** İstanbul UTC+3 (2016'dan beri yaz saati yok). */
const ISTANBUL_OFFSET_MS = 3 * 36e5

/** İstanbul saatine göre bu ayın ilk anı (UTC). */
export function monthStart(now: Date): Date {
  const local = new Date(now.getTime() + ISTANBUL_OFFSET_MS)
  return new Date(Date.UTC(local.getUTCFullYear(), local.getUTCMonth(), 1) - ISTANBUL_OFFSET_MS)
}

/** Kotanın yenileneceği an: İstanbul'da gelecek ayın 1'i 00:00. */
export function nextMonthStart(now: Date): Date {
  const local = new Date(now.getTime() + ISTANBUL_OFFSET_MS)
  return new Date(Date.UTC(local.getUTCFullYear(), local.getUTCMonth() + 1, 1) - ISTANBUL_OFFSET_MS)
}

/**
 * Bir özelliğin kullanılıp kullanılamayacağı (spec §4.2). Abonelikte kota yok,
 * yalnızca uyarlamada günlük tavan; ücretsizde aylık kota, abonelik
 * özellikleri kapalı.
 */
export function decide(feature: Feature, input: { subs: SubscriptionLike[]; usage: Usage; now: Date; unlimited?: boolean }): Decision {
  if (input.unlimited) return { allowed: true, via: "admin" }
  if (hasAccess(input.subs, input.now)) {
    if (feature === "adaptation" && input.usage.adaptationsLast24h >= DAILY_CAP) return { allowed: false, code: "daily_cap" }
    return { allowed: true, via: "subscription" }
  }
  if (feature === "analysis") {
    return input.usage.analysesThisMonth < FREE_QUOTA.analyses ? { allowed: true, via: "free" } : { allowed: false, code: "analysis_quota" }
  }
  if (feature === "adaptation") {
    return input.usage.adaptationsThisMonth < FREE_QUOTA.adaptations ? { allowed: true, via: "free" } : { allowed: false, code: "adaptation_quota" }
  }
  return { allowed: false, code: "subscription_required" }
}
```

```ts
export async function requireEntitlement(db: Client, input: { userId: string; feature: Feature; now: Date; unlimited?: boolean }): Promise<void> {
```

```ts
export interface PolarDeps {
  createCheckout(body: { products: string[]; external_customer_id: string; customer_email: string; success_url: string; metadata: Record<string, string>; locale: string }): Promise<{ url: string }>
  createCustomerSession(body: { external_customer_id: string; return_url: string }): Promise<{ customer_portal_url: string }>
  revokeSubscription(id: string): Promise<unknown>
  validateEvent(body: string, headers: Record<string, string>, secret: string): Promise<{ type: string; data: unknown }>
  isVerificationError(error: unknown): boolean
  isUnknownTypeError(error: unknown): boolean
}

/** Abonelik durumunu değiştiren bildirimler; hepsi aynı yere gidiyor (spec §4.3). */
const SUBSCRIPTION_EVENTS = new Set([
  "subscription.created", "subscription.updated", "subscription.active", "subscription.canceled",
  "subscription.uncanceled", "subscription.revoked", "subscription.past_due",
])

interface SubscriptionData {
  id?: unknown; status?: unknown; product_id?: unknown; current_period_end?: unknown; cancel_at_period_end?: unknown
  trial_end?: unknown; ended_at?: unknown; created_at?: unknown; modified_at?: unknown
  metadata?: Record<string, unknown> | null; customer?: { external_id?: unknown } | null
}

const date = (v: unknown) => (typeof v === "string" && v ? new Date(v) : null)

export function createPolarProvider(config: { webhookSecret: string; products: Partial<Record<"month" | "quarter", string>> }, deps: PolarDeps): BillingProvider {
  const planOf = (productId: unknown) =>
    (Object.entries(config.products).find(([, id]) => id && id === productId)?.[0] as "month" | "quarter" | undefined) ?? null
  return {
    async createCheckout({ userId, email, productId, plan, successUrl }) {
      const checkout = await deps.createCheckout({
        products: [productId], external_customer_id: userId, customer_email: email, success_url: successUrl,
        metadata: { plan, user_id: userId }, locale: "tr",
      })
      return { url: checkout.url }
    },
    async createPortalSession({ userId, returnUrl }) {
      const session = await deps.createCustomerSession({ external_customer_id: userId, return_url: returnUrl })
      return { url: session.customer_portal_url }
    },
    async revokeSubscription(id) {
      await deps.revokeSubscription(id)
    },
    async parseWebhook(rawBody, headers): Promise<BillingEvent | null> {
      let event: { type: string; data: unknown }
      try {
        event = await deps.validateEvent(rawBody, headers, config.webhookSecret)
      } catch (error) {
        if (deps.isVerificationError(error)) throw new WebhookSignatureError()
        if (deps.isUnknownTypeError(error)) return null
        throw error
      }
      if (!SUBSCRIPTION_EVENTS.has(event.type)) return null
      const s = event.data as SubscriptionData
      const userId = s.metadata?.user_id || s.customer?.external_id
      const plan = planOf(s.product_id)
      const modifiedAt = date(s.modified_at) ?? date(s.created_at)
      if (typeof s.id !== "string" || typeof userId !== "string" || !userId || !plan || typeof s.status !== "string" || !modifiedAt) return null
      return {
        type: "subscription", subscriptionId: s.id, userId, plan, status: s.status,
        currentPeriodEnd: date(s.current_period_end), cancelAtPeriodEnd: s.cancel_at_period_end === true,
        trialEnd: date(s.trial_end), endedAt: date(s.ended_at), modifiedAt,
      }
    },
  }
}
```

```ts
import * as Sentry from "@sentry/nextjs"
import type { prisma } from "@uyarla/db"
import { grantsAccess } from "@/features/billing/access"
import type { BillingProvider, SubscriptionEvent } from "@/server/billing/provider"

type Db = typeof prisma

/**
 * Abonelik bildirimini uygular (DOG-158): Polar'daki son hâle ekle-ya da-güncelle.
 * Polar bildirimleri sırasız gönderebiliyor; kayıttakinden eski `modifiedAt`
 * yok sayılıyor ("stale"), eşit olan yeniden yazılıyor (aynı bildirim iki kez).
 * Kullanıcı yoksa (hesap silinmiş) "ignored": çağıran aboneliği Polar'da iptal ediyor.
 */
export async function applySubscriptionEvent(db: Db, event: SubscriptionEvent): Promise<"saved" | "stale" | "ignored"> {
  const user = await db.user.findUnique({ where: { id: event.userId }, select: { id: true } })
  if (!user) return "ignored"
  const data = {
    plan: event.plan,
    status: event.status,
    currentPeriodEnd: event.currentPeriodEnd,
    cancelAtPeriodEnd: event.cancelAtPeriodEnd,
    trialEnd: event.trialEnd,
    endedAt: event.endedAt,
    providerUpdatedAt: event.modifiedAt,
  }
  return db.$transaction(async (tx) => {
    // Aynı aboneliğin eşzamanlı iki bildirimi sırayla.
    await tx.$queryRaw`SELECT id FROM "User" WHERE id = ${user.id} FOR UPDATE`
    const existing = await tx.subscription.findUnique({ where: { providerSubscriptionId: event.subscriptionId }, select: { providerUpdatedAt: true } })
    if (existing && existing.providerUpdatedAt > event.modifiedAt) return "stale" as const
    await tx.subscription.upsert({
      where: { providerSubscriptionId: event.subscriptionId },
      create: { userId: user.id, providerSubscriptionId: event.subscriptionId, ...data },
      update: data,
    })
    return "saved" as const
  })
}

/**
 * Hesap silinirken açık abonelikleri Polar'da iptal eder; yoksa silinmiş
 * hesaptan ücret çekilmeye devam ederdi. Polar'a ulaşılamazsa silme durmuyor:
 * uyarı düşüyor, iptal elle yapılıyor (spec §4.3). İptal edilen sayıyı döndürür.
 */
export async function revokeOpenSubscriptions(
  db: Db,
  provider: BillingProvider,
```

```ts
export async function handleWebhook(input: { db: Db; provider: BillingProvider; rawBody: string; headers: Record<string, string>; now: Date; log?: (message: string, extra: unknown) => void }): Promise<Reply> {
  const log = input.log ?? ((message: string, extra: unknown) => {
    console.warn(message, extra)
    Sentry.captureMessage(message, { level: "warning", extra: { event: extra } })
  })
  let event
  try {
    event = await input.provider.parseWebhook(input.rawBody, input.headers)
  } catch (error) {
    if (error instanceof WebhookSignatureError) return { status: 403, json: { error: "invalid_signature" } }
    throw error
  }
  if (!event) return { status: 200, json: { received: true } }
  const outcome = await applySubscriptionEvent(input.db, event)
  if (outcome === "ignored") {

    await input.provider.revokeSubscription(event.subscriptionId).catch((error: unknown) => log("[billing] abonelik iptal edilemedi", { id: event.subscriptionId, error: String(error) }))
  }
  return { status: 200, json: { received: true, outcome } }
}
```

```ts
export type PlanKey = "month" | "quarter"

export const PLAN_MONTHS: Record<PlanKey, number> = { month: 1, quarter: 3 }

const ORDER: PlanKey[] = ["month", "quarter"]
const PRODUCT_ENV: Record<PlanKey, string> = { month: "POLAR_PRODUCT_MONTH", quarter: "POLAR_PRODUCT_QUARTER" }

export function isPlanKey(value: unknown): value is PlanKey {
  return typeof value === "string" && (ORDER as string[]).includes(value)
}

export interface PlanOffer {
  key: PlanKey
  months: number
  productId: string
  /** Fatura dönemi başına TL; ödeme sayfasındaki tutar Polar'dan. */
  priceTry: number | null
}

/** Ürün kimliği tanımlı paketler; `PLAN_PRICES_TRY` "month:299, quarter:599". */
export function plansFromEnv(env: Record<string, string | undefined>): PlanOffer[] {
  const prices = new Map(
    (env.PLAN_PRICES_TRY ?? "")
      .split(",")
      .map((pair) => pair.split(":").map((p) => p.trim()))
      .filter(([key, price]) => key && price && Number.isFinite(Number(price)))
      .map(([key, price]) => [key!, Number(price)]),
  )
  return ORDER.flatMap((key) => {
    const productId = env[PRODUCT_ENV[key]]?.trim()
    return productId ? [{ key, months: PLAN_MONTHS[key], productId, priceTry: prices.get(key) ?? null }] : []
  })
}

/** Aylık karşılığı, tam liraya yuvarlanmış (599/3 → 200). */
export function monthlyPrice(plan: PlanOffer): number | null {
  return plan.priceTry === null ? null : Math.round(plan.priceTry / plan.months)
}

/** 3 Ay'ın 1 Ay'a göre tasarrufu, yüzde; ikisinin de fiyatı yoksa null. */

  const quarter = plans.find((p) => p.key === "quarter")
  if (!month?.priceTry || !quarter?.priceTry) return null
  return Math.round((1 - quarter.priceTry / quarter.months / month.priceTry) * 100)
```

```ts
import type { BillingStatus } from "@/features/billing/api"
import { formatDay, timeLeft } from "@/features/billing/format"

const URGENT_MS = 2 * 864e5

/** Uygulamanın üstündeki deneme şeridi (spec §5); denemede değilse null. */
export function trialBanner(status: BillingStatus, now: Date): { text: string; urgent: boolean } | null {
  const sub = status.subscription
  if (!sub || sub.status !== "trialing" || !sub.trialEnd) return null
  const urgent = new Date(sub.trialEnd).getTime() - now.getTime() <= URGENT_MS
  if (sub.cancelAtPeriodEnd) return { text: `Deneme ${formatDay(sub.trialEnd)} tarihinde bitiyor; ücret çekilmeyecek`, urgent }
  const charge = sub.priceTry !== null ? `${sub.priceTry} ₺` : "paket ücreti"
  return { text: `Deneme: ${timeLeft(sub.trialEnd, now)} · ${formatDay(sub.trialEnd)} tarihinde ${charge} çekilecek`, urgent }
}
```
